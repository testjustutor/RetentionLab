3. JAVASCRIPT TABLE RENDERING FUNCTIONS

Replace chart rendering functions with table rendering functions:

function renderYourTable(data) {
  const tbody = document.getElementById('tableId');
  if (!tbody) return;

  if (!data || !data.labels || !data.data) {
    tbody.innerHTML = '<tr><td colspan="3" class="py-2 text-center text-[color]-800 font-medium">No data available</td></tr>';
    return;
  }

  const total = data.data.reduce((sum, count) => sum + count, 0);
  
  tbody.innerHTML = data.labels.map((label, index) => {
    const count = data.data[index] || 0;
    const percentage = total > 0 ? ((count / total) * 100).toFixed(1) : 0;
    
    return `
      <tr class="border-b border-[color]-300 hover:bg-[color]-100 transition-colors">
        <td class="py-2 px-2 text-[11px] font-bold text-[color]-950">${escapeHtml(label)}</td>
        <td class="py-2 px-2 text-[11px] font-bold text-[color]-900 text-right">${count}</td>
        <td class="py-2 px-2 text-[11px] font-semibold text-[color]-700 text-right">${percentage}%</td>
      </tr>
    `;
  }).join('');
}

4. TYPOGRAPHY STANDARDS

Headers:

- Font weight: font-bold
- Style: uppercase tracking-wide
- Color: Use the darkest shade of your theme color (e.g., text-emerald-900)

Body Text:
- Font size: text-[11px]
- Font weight: font-semibold or font-bold
- Color: Use 950 for primary data, 900 for secondary, 700 for tertiary

Table Headers:
- Font size: text-[10px]
- Font weight: font-bold
- Style: uppercase
- Color: text-[color]-950 (darkest shade)
